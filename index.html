  <p id="info">Loading…</p>

  <button id="btn" class="hidden">Email me the report</button>

  <div id="msg"></div>

</div>

<script>

  const EXEC_URL = 'https://script.google.com/macros/s/AKfycby1qWG4SxKBRfxv7FUY1-kjBMxbaDIY-jYtercnipBAPLZBvtQ-ZH6na8ASfWvj9YRy/exec';

  const token = new URLSearchParams(location.search).get('t') || '';

  const $ = id => document.getElementById(id);


  function call(action) {

    return fetch(EXEC_URL, {

      method: 'POST',

      credentials: 'omit',                       // no Google cookies -> no multi-account bug

      headers: { 'Content-Type': 'text/plain' }, // avoids CORS preflight

      body: JSON.stringify({ action, t: token })

    }).then(r => r.json());

  }


  call('info').then(r => {

    if (!r.ok) { $('title').textContent = 'Link unavailable'; $('info').textContent = r.msg; return; }

    $('info').innerHTML = '';

    $('info').append('Report for ', Object.assign(document.createElement('b'), { textContent: r.student }),

      '. Click below and the PDF will be emailed to ', Object.assign(document.createElement('b'), { textContent: r.masked }), '.');

    $('btn').classList.remove('hidden');

  }).catch(() => { $('info').textContent = 'Could not load. Please try again later.'; });


  $('btn').onclick = () => {

    $('btn').disabled = true; $('msg').textContent = 'Preparing your report…';

    call('send').then(r => { $('msg').textContent = r.msg; if (!r.ok) $('btn').disabled = false; })

      .catch(() => { $('msg').textContent = 'Something went wrong. Please try again.'; $('btn').disabled = false; });

  };

</script>
